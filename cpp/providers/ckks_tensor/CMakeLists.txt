# Native tensor CKKS implementation shared by the model SDK and Scope adapter.
add_library(fhn_ckks_tensor SHARED engine.cpp transforms.cpp refresh.cpp backend.cpp)
target_compile_features(fhn_ckks_tensor PRIVATE cxx_std_17)
target_include_directories(fhn_ckks_tensor PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>)
target_link_libraries(fhn_ckks_tensor PRIVATE fhn_ckks_keys fhn_ckks_eval nlohmann_json::nlohmann_json)
set_target_properties(fhn_ckks_tensor PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
install(TARGETS fhn_ckks_tensor LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES backend.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/FHN RENAME fhn_tensor_provider.h)

if(FHN_BUILD_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(FHN_CHEDDAR_ROOT "${FHENOMENON_REPOSITORY_ROOT}/refs/cheddar-fhe" CACHE PATH "Pinned Cheddar source")
  if(NOT EXISTS "${FHN_CHEDDAR_ROOT}/include/common/Basic.cuh")
    message(FATAL_ERROR "CUDA requires the Cheddar submodule: git submodule update --init refs/cheddar-fhe")
  endif()
  target_sources(fhn_ckks_tensor PRIVATE cuda/CudaEngine.cu cuda/CudaPrimitives.cu)
  target_include_directories(fhn_ckks_tensor PRIVATE "${FHN_CHEDDAR_ROOT}/include")
  target_compile_definitions(fhn_ckks_tensor PRIVATE FHN_NATIVE_CUDA=1)
  target_link_libraries(fhn_ckks_tensor PRIVATE CUDA::cudart)
  set_target_properties(fhn_ckks_tensor PROPERTIES CUDA_STANDARD 17)
  install(FILES cuda/LICENSE.cheddar DESTINATION ${CMAKE_INSTALL_DATADIR}/Fhenomenon/licenses)
endif()
